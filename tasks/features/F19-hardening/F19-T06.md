### Test groups

#### 1. data-io tests

```ts
import { describe, test, expect } from "bun:test";
import { readJson, readJsonSafe, validateJson, DataCorruptionError } from "../lib/data-io.js";
import { writeFileSync, unlinkSync, mkdirSync } from "fs";
import { join } from "path";
import { tmpdir } from "os";

describe("readJson", () => {
  const TMP = join(tmpdir(), "f19-test");

  // Setup: create temp dir
  // Teardown: remove temp dir

  test("throws DataCorruptionError with conflict_markers for file with git markers", () => {
    const path = join(TMP, "conflict.json");
    writeFileSync(path, '[\n<<<<<<< HEAD\n{"a":1}\n=======\n{"b":2}\n>>>>>>> origin/main\n]');
    expect(() => readJson(path)).toThrow(DataCorruptionError);
    // Check reason
  });

  test("throws DataCorruptionError with invalid_json for malformed JSON", () => {
    const path = join(TMP, "bad.json");
    writeFileSync(path, "{not valid json}}}");
    expect(() => readJson(path)).toThrow(DataCorruptionError);
  });

  test("throws DataCorruptionError with file_not_found for missing file", () => {
    expect(() => readJson(join(TMP, "nope.json"))).toThrow(DataCorruptionError);
  });

  test("parses valid JSON successfully", () => {
    const path = join(TMP, "good.json");
    writeFileSync(path, '[{"name":"test"}]');
    const data = readJson<Array<{name: string}>>(path);
    expect(data).toEqual([{name: "test"}]);
  });
});

describe("readJsonSafe", () => {
  test("returns fallback for corrupted file", () => {
    const path = join(TMP, "corrupt.json");
    writeFileSync(path, "<<<<<<< HEAD\nnot json");
    const result = readJsonSafe(path, []);
    expect(result).toEqual([]);
  });
});

describe("validateJson", () => {
  test("returns false for file with conflict markers", ...);
  test("returns false for invalid JSON", ...);
  test("returns true for valid JSON", ...);
});
```
